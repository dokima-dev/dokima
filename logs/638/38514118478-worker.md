> LINK = re.compile(r"\]\(([^)#\s:]+\.md)(?:#[^)]*)?\)")

> def rules_files(root=ROOT):
>     """`AGENTS.md`, then every existing repo markdown file it links, as repo paths."""
>     index = open(os.path.join(root, "AGENTS.md"), encoding="utf-8").read()
>     out = ["AGENTS.md"]
